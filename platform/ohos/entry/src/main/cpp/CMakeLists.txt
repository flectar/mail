cmake_minimum_required(VERSION 3.16)
project(feihong LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FEIHONG_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
if(DEFINED ENV{FEIHONG_RUST_LIBRARY})
  set(FEIHONG_RUST_LIBRARY "$ENV{FEIHONG_RUST_LIBRARY}")
else()
  set(FEIHONG_RUST_LIBRARY "${FEIHONG_REPO_ROOT}/target/aarch64-unknown-linux-ohos/release/libfeihong_rust.a")
endif()
if(NOT EXISTS "${FEIHONG_RUST_LIBRARY}")
  message(FATAL_ERROR "Rust archive missing: ${FEIHONG_RUST_LIBRARY}. Run scripts/ohos-build.py first.")
endif()
add_library(feihong_rust STATIC IMPORTED)
set_target_properties(feihong_rust PROPERTIES IMPORTED_LOCATION "${FEIHONG_RUST_LIBRARY}")
set(FEIHONG_OPENSSL_ROOT "${FEIHONG_REPO_ROOT}/target/ohos/openssl/arm64-v8a")
foreach(component ssl crypto)
  set(archive "${FEIHONG_OPENSSL_ROOT}/lib/lib${component}.a")
  if(NOT EXISTS "${archive}")
    message(FATAL_ERROR "OHOS OpenSSL archive missing: ${archive}. Run scripts/ohos-openssl.py first.")
  endif()
  add_library(feihong_${component} STATIC IMPORTED)
  set_target_properties(feihong_${component} PROPERTIES IMPORTED_LOCATION "${archive}")
endforeach()
add_library(feihong SHARED bridge.cpp graphics.cpp input.cpp)
set_target_properties(feihong PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
target_include_directories(feihong PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(feihong PRIVATE -Wall -Wextra -Werror=return-type)
# Keep debug assertions/symbols while avoiding the SDK's default -O0 in the
# presentation and input bridge. Last optimization flag wins in Clang.
target_compile_options(feihong PRIVATE $<$<CONFIG:Debug>:-O2>)
target_link_options(feihong PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,--exclude-libs,ALL)
target_link_libraries(feihong PRIVATE -Wl,--start-group feihong_rust feihong_ssl feihong_crypto
  -Wl,--end-group ace_napi.z ace_ndk.z native_window
  native_buffer native_vsync ohinputmethod hilog_ndk.z native_drawing asset_ndk.z pasteboard udmf
  time_service_ndk qos uv dl m)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(feihong PRIVATE FEIHONG_DEBUG_TESTS=1)
  target_sources(feihong PRIVATE "${FEIHONG_REPO_ROOT}/platform/ohos/tests/pdfium_probe.c")
  target_include_directories(feihong PRIVATE "${FEIHONG_REPO_ROOT}/target/pdfium-worktree/pdfium")
  add_library(feihong_pdfium SHARED IMPORTED)
  set_target_properties(feihong_pdfium PROPERTIES IMPORTED_LOCATION
    "${FEIHONG_REPO_ROOT}/platform/ohos/entry/libs/arm64-v8a/libpdfium.so")
  target_link_libraries(feihong PRIVATE feihong_pdfium)
endif()
